cmake_minimum_required(VERSION 3.24)
project(KiraRaki VERSION 0.2.0 LANGUAGES C CXX OBJC OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0)
set(CMAKE_OSX_ARCHITECTURES arm64)
set(CMAKE_OBJCXX_FLAGS "${CMAKE_OBJCXX_FLAGS} -fobjc-arc")
add_library(KiraCore STATIC core/Contours.cpp)
target_include_directories(KiraCore PUBLIC core)
add_library(KiraEngine STATIC engine/KREngine.mm engine/KRMedia.mm shared/KRIPC.mm)
target_include_directories(KiraEngine PUBLIC engine shared core)
target_link_libraries(KiraEngine PUBLIC KiraCore "-framework Foundation" "-framework AppKit" "-framework Metal" "-framework MetalKit" "-framework IOSurface" "-framework CoreVideo" "-framework CoreMedia" "-framework CoreImage" "-framework AVFoundation" "-framework Vision" "-framework CoreML" "-framework Accelerate" "-framework Security")
add_executable(KiraRakiService service/main.mm)
target_link_libraries(KiraRakiService PRIVATE KiraEngine)
add_executable(KiraRakiCLI cli/main.mm)
target_link_libraries(KiraRakiCLI PRIVATE KiraEngine)
option(KIRA_BUILD_OBS "Build the OBS filter and source" OFF)
if(KIRA_BUILD_OBS)
  add_subdirectory(obs)
endif()
