set(KIRA_OBS_SOURCE "" CACHE PATH "OBS 32.2.2 source checkout (headers only)")
set(KIRA_OBS_APP "/Applications/OBS.app" CACHE PATH "OBS application to link against")
if(NOT EXISTS "${KIRA_OBS_SOURCE}/libobs/obs.h")
  message(FATAL_ERROR "Set KIRA_OBS_SOURCE to an OBS 32.2.2 source checkout")
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/obsconfig.h" "#pragma once\n#define OBS_DATA_PATH \"\"\n#define OBS_PLUGIN_PATH \"\"\n#define OBS_INSTALL_PREFIX \"\"\n#define OBS_RELEASE_CANDIDATE 0\n#define OBS_BETA 0\n")
add_library(kiraraki MODULE plugin.mm)
target_include_directories(kiraraki PRIVATE "${KIRA_OBS_SOURCE}/libobs" "${CMAKE_CURRENT_BINARY_DIR}/generated" "${PROJECT_SOURCE_DIR}/.deps/simde")
target_link_libraries(kiraraki PRIVATE KiraEngine "${KIRA_OBS_APP}/Contents/Frameworks/libobs.framework/Versions/A/libobs")
set_target_properties(kiraraki PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION plugin PREFIX "" BUILD_RPATH "${KIRA_OBS_APP}/Contents/Frameworks" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist")
add_executable(KiraRakiOBSCheck EXCLUDE_FROM_ALL quick-check.mm)
target_include_directories(KiraRakiOBSCheck PRIVATE "${KIRA_OBS_SOURCE}/libobs" "${CMAKE_CURRENT_BINARY_DIR}/generated" "${PROJECT_SOURCE_DIR}/.deps/simde")
target_link_libraries(KiraRakiOBSCheck PRIVATE "${KIRA_OBS_APP}/Contents/Frameworks/libobs.framework/Versions/A/libobs" "-framework AppKit")
set_target_properties(KiraRakiOBSCheck PROPERTIES BUILD_RPATH "${KIRA_OBS_APP}/Contents/Frameworks")
